import { Meta, ColorPalette, ColorItem, Typeset } from "@storybook/blocks";
import { TabsState } from "@storybook/components";
import { lightTheme, darkTheme } from "../styles/theme";

<Meta title="Design System/Colors" />

export const sortObjectKeys = (obj) =>
  Object.keys(obj)
    .sort()
    .reduce((acc, key) => {
      acc[key] = obj[key];
      return acc;
    }, {});

export const getColors = (theme) =>
  Object.keys(sortObjectKeys(theme.color)).map((key) => ({
    name: key,
    value: theme.color[key],
  }));

# Colors

<TabsState initial="light">
  <div id="light" title="Light Mode">
    <ColorPalette>
      {getColors(lightTheme).map(({ name, value }) => (
        <ColorItem
          key={name}
          title={`theme.colors.${name}`}
          subtitle={name}
          colors={[value]}
        />
      ))}
    </ColorPalette>
  </div>
  <div id="dark" title="Dark Mode">
    <ColorPalette>
      {getColors(darkTheme).map(({ name, value }) => (
        <ColorItem
          key={name}
          title={`theme.colors.${name}`}
          subtitle={name}
          colors={[value]}
        />
      ))}
    </ColorPalette>
  </div>
</TabsState>
